      :document-server-url="serverUrl"
      :config="mergedConfig"

<script setup>
import { ref, computed, onUnmounted } from 'vue'
import { DocumentEditor } from '@onlyoffice/document-editor-vue'

const TAG = '[OO调试]'

const props = defineProps({
  id: { type: String, default: () => `onlyoffice-editor-${Date.now()}` },
  serverUrl: { type: String, required: true },
  config: {
    type: Object,
    required: true,
    validator: (val) => !!(val.document?.url && val.document?.fileType && val.document?.key)
  },
  mode: { type: String, default: 'edit', validator: (v) => ['edit', 'view'].includes(v) },
  lang: { type: String, default: 'zh-CN' }
})

const emit = defineEmits(['ready', 'error', 'request-close'])

const isDirty = ref(false)
const isReady = ref(false)
const editorInstance = ref(null)

let pendingSaveResolve = null
let pendingSaveTimer = null

// ===== 诊断①：直接监听编辑器 iframe 的 postMessage，确认编辑器到底有没有发事件 =====
const onWindowMessage = (e) => {
  const d = e.data
  if (d && typeof d === 'object' && typeof d.type === 'string' && /^on[A-Z]/.test(d.type)) {
    console.log(TAG, 'iframe 事件 =>', d.type, d)
  }
}
window.addEventListener('message', onWindowMessage)

const clearPendingSave = (result) => {
  if (pendingSaveTimer) { clearTimeout(pendingSaveTimer); pendingSaveTimer = null }
  if (pendingSaveResolve) {
    const r = pendingSaveResolve
    pendingSaveResolve = null
    r(result)
  }
}

function onReady() {
  editorInstance.value = window.DocEditor?.instances?.[props.id] || null
  isReady.value = true
  isDirty.value = false
  console.log(TAG, 'onDocumentReady 触发，实例=', editorInstance.value,
    '可用方法=', editorInstance.value ? Object.keys(editorInstance.value) : null)
  emit('ready')
}

function onStateChange(e) {
  console.log(TAG, 'onDocumentStateChange 触发，payload=', e)
  const changed = !!e?.data
  if (changed) {
    isDirty.value = true                        // Fast 模式下 latch：一旦改过就保持脏，直到主动保存
  } else if (pendingSaveResolve) {
    isDirty.value = false
    clearPendingSave(true)
  }
  console.log(TAG, 'isDirty =', isDirty.value)
}

function onEditorError(err) {
  console.error(TAG, 'onError =', err)
  if (pendingSaveResolve) clearPendingSave(false)
  emit('error', err)
}

function onRequestClose() {
  console.log(TAG, 'onRequestClose 触发')
  emit('request-close')
}

const events = {
  onAppReady: () => console.log(TAG, 'onAppReady 触发'),
  onDocumentReady: onReady,
  onDocumentStateChange: onStateChange,
  onError: onEditorError,
  onRequestClose
}

const mergedConfig = computed(() => {
  const cfg = {
    ...props.config,
    editorConfig: {
      callbackUrl: "https://nygl/onlyOffice/callback",
      lang: props.lang,
      ...(props.mode === 'view' ? { mode: props.mode } : {}),
      customization: {
        ...props.config.editorConfig?.customization,
        forcesave: true,
        autosave: false
      },
      user: {
        ...props.config.editorConfig?.user,
        name: props.config.editorConfig?.user?.name || 'Anonymous'
      }
    },
    events
  }
  // ===== 诊断②：确认交给封装库的配置里确实带了事件 =====
  console.log(TAG, 'mergedConfig 生成，events=', Object.keys(cfg.events), 'mode=', props.mode, 'token=', !!cfg.token)
  return cfg
})

function handleSave() {
  const inst = editorInstance.value
  if (!inst) return Promise.resolve(false)
  if (!isDirty.value) return Promise.resolve(true)

  return new Promise((resolve) => {
    pendingSaveResolve = resolve
    pendingSaveTimer = setTimeout(() => clearPendingSave(false), 10000)
    try {
      inst.forceSave()
    } catch (e) {
      console.error(TAG, 'forceSave 异常', e)
      clearPendingSave(false)
    }
  })
}

// 路线B（Strict 模式）使用：触发编辑器原生未保存确认框
function requestClose() {
  const inst = editorInstance.value || window.DocEditor?.instances?.[props.id]
  console.log(TAG, 'requestClose 调用，实例=', inst)
  if (inst && typeof inst.requestClose === 'function') {
    inst.requestClose()
  } else {
    emit('request-close')
  }
}

onUnmounted(() => {
  window.removeEventListener('message', onWindowMessage)
  clearPendingSave(false)
  try { editorInstance.value?.destroyEditor?.() } catch {}
  if (window.DocEditor?.instances) window.DocEditor.instances[props.id] = undefined
  editorInstance.value = null
})

defineExpose({
  isDirty,
  isReady,
  getIsDirty: () => isDirty.value,
  save: handleSave,
  requestClose
})
</script>

// Fast 模式：以宿主维护的脏标记为准
const confirmClose = async () => {
  const editor = editorRef.value
  console.log('[关闭诊断] editorRef=', editor, 'isReady=', editor?.isReady, 'isDirty=', editor?.getIsDirty?.())

  if (!editor || !editor.isReady) return true          // 编辑器还没加载完，不拦
  if (!editor.getIsDirty()) return true                // 没改过，直接关

  try {
    await ElMessageBox.confirm(
      '文档有未保存的更改，是否保存后再关闭？',
      '提示',
      {
        confirmButtonText: '保存并关闭',
        cancelButtonText: '不保存',
        distinguishCancelAndClose: true,
        type: 'warning'
      }
    )
    isSaving.value = true
    const ok = await editor.save()
    isSaving.value = false
    if (!ok) {
      ElMessage.error('保存失败，请检查回调服务后重试')
      return false
    }
    return true
  } catch (action) {
    return action === 'cancel'                         // 「不保存」放行；X/ESC 留下
  }
}
