      :document-server-url="serverUrl"

<script setup>
import { ref, computed, onUnmounted } from 'vue'
import { DocumentEditor } from '@onlyoffice/document-editor-vue'

const props = defineProps({
  id: { type: String, default: () => `onlyoffice-editor-${Date.now()}` },
  serverUrl: { type: String, required: true },
  config: {
    type: Object,
    required: true,
    validator: (val) => !!(val.document?.url && val.document?.fileType && val.document?.key)
  },
  mode: { type: String, default: 'edit', validator: (v) => ['edit', 'view'].includes(v) },
  lang: { type: String, default: 'zh-CN' }
})

const emit = defineEmits(['ready', 'error'])

// 是否有未保存的修改（由 onDocumentStateChange 维护）
const isDirty = ref(false)
const editorInstance = ref(null)

// 等待 forceSave 完成的 resolver + 兜底定时器
let pendingSaveResolve = null
let pendingSaveTimer = null

const clearPendingSave = (result) => {
  if (pendingSaveTimer) { clearTimeout(pendingSaveTimer); pendingSaveTimer = null }
  if (pendingSaveResolve) {
    const r = pendingSaveResolve
    pendingSaveResolve = null
    r(result)
  }
}

// 编辑器事件：放进 config.events 才会真正生效
const events = {
  onDocumentReady: onReady,
  onDocumentStateChange: onStateChange,
  onError: onEditorError
}

const mergedConfig = computed(() => ({
  ...props.config,
  editorConfig: {
    // ...props.config.editorConfig,
    // 注意：callbackUrl 必须是后端真实可达的地址，否则 forceSave 不会落盘
    callbackUrl: "https://nygl/onlyOffice/callback",
    lang: props.lang,
    ...(props.mode === 'view' ? { mode: props.mode } : {}),
    // mode: "edit",
    customization: {
      ...props.config.editorConfig?.customization,
      forcesave: true,   // 启用强制保存
      autosave: false   // 关闭自动保存
    },
    user: {
      ...props.config.editorConfig?.user,
      name: props.config.editorConfig?.user?.name || 'Anonymous'
    }
  },
  events  // 放在 spread 之后，确保覆盖父组件可能误传的 events
}))

function onReady() {
  // onDocumentReady 触发时实例已注册（见 DocumentEditor.vue 第 307-308 行）
  editorInstance.value = window.DocEditor?.instances?.[props.id] || null
  isDirty.value = false
  emit('ready')
}

// function onStateChange(e) {
//   const changed = !!e?.data
//   isDirty.value = changed
//   // forceSave 成功后，编辑服务会回推 data:false —— 此时判定保存完成
//   if (!changed && pendingSaveResolve) clearPendingSave(true)
// }
function onStateChange(e) {
  const changed = !!e?.data
  if (changed) {
    isDirty.value = true            // 一旦改过就置脏
  } else if (pendingSaveResolve) {
    isDirty.value = false           // 只有我们主动保存完成才清脏
    clearPendingSave(true)
  }
  // 其它 data:false（自动保存/协同/初始加载）一律不动脏标记
}

function onEditorError(err) {
  if (pendingSaveResolve) clearPendingSave(false) // 保存中断，结束等待
  emit('error', err)
}

function onDestroy() {
  editorInstance.value = null
  isDirty.value = false
}

// 返回 Promise：true=保存成功 / 无需保存；false=失败或无实例
function handleSave() {
  const inst = editorInstance.value
  if (!inst) return Promise.resolve(false)
  if (!isDirty.value) return Promise.resolve(true) // 没改过，直接算成功

  return new Promise((resolve) => {
    pendingSaveResolve = resolve
    pendingSaveTimer = setTimeout(() => clearPendingSave(false), 10000) // 兜底，防弹框卡死
    try {
      inst.forceSave()
    } catch (e) {
      console.error('保存失败', e)
      clearPendingSave(false)
    }
  })
}

onUnmounted(() => {
  clearPendingSave(false)
  try { editorInstance.value?.destroyEditor() } catch {}
  if (window.DocEditor?.instances) window.DocEditor.instances[props.id] = undefined
  editorInstance.value = null
})

defineExpose({
  isDirty,                  // 父组件可直接当 boolean 读（Vue 会自动解包 ref）
  getIsDirty: () => isDirty.value,
  save: handleSave
})
</script>

    }

const submitForm = () => {
    if(submitting.value) return // 防重复提交
    submitting.value = true

  // 1. 表单字段校验
  // const valid = suggestFormRef.value?.validate().catch(() => false)
  // if (!valid) return

    suggestFormRef.value?.validate(async (valid: boolean) => {
    if (valid) {
      let hrefId = Date.now().toString()
      if(!form.value.id) {
        form.value.hrefId = hrefId // 新增
      } else {
        hrefId = form.value.id // 修改
      }
      // 处理附件（统一上传附件）
      // 2. 附件校验：至少有一个文件
      if (fileList.value.length === 0) {
          ElMessage.warning('请至少选择一个附件')
          submitting.value = false
          return
      }
      //2. 校验每个文件的类型和大小
      // for (const fileItem of fileList.value) {
      //     // fileItem.raw 是原始 File 对象
      //     if (!validateFile(fileItem.raw)) {
      //         return
      //     }
      // }

      try {
          // 3. 附件增删处理（修改建议）
          // 删除用户移除的附件
          if (deleteAttachmentIds.value.length > 0) {
            //console.log("要删除的文件：", deleteAttachmentIds.value)
            // 批量删除附件
            await delProfile(deleteAttachmentIds.value);
          }

          // 只上传【新选择的文件】（老文件不上传，只有新文件才有raw）
          const newFiles = [...fileList.value].filter(item => item.raw);
          //4. 校验每个文件的类型和大小
          for (const fileItem of newFiles) {
            // fileItem.raw 是原始 File 对象
            if (!validateFile(fileItem)) {
              submitting.value = false
              return
            }
          }

          // 5. 依次上传所有附件
          for (const fileItem of newFiles) {
              // console.log(fileItem.raw)
              await new Promise(resolve => setTimeout(resolve, 600)) // 为了避开拦截器，短时间内同一个接口的请求太多可能会被拦截
              // const fileKey = attachment_type.value.find(item => item.label === fileItem.raw.type)?.value;
              const attachmentformData = new FormData()
              attachmentformData.append('file', fileItem.raw)
              attachmentformData.append('fileName', fileItem.raw.name)
              attachmentformData.append('hrefId', hrefId)
              attachmentformData.append('hrefType', hrefType.value) // fileKey

            //console.log("上传前：", fileItem.raw.uid)
              const res = await uploadAttachment(attachmentformData)

            // const formDataFile = new FormData()
              // formDataFile.append('file', fileItem.raw)
              // const res = await uploadFile(formDataFile)
              if (res.code === 200 && res.data) {
                  const fileUrl = prefix + res.data.href
                  const fileName = res.data.fileName
                  fileItem.raw.url = fileUrl
                  fileItem.raw.id = res.data.id
                  // uploadedUrls.push(fileUrl)
                  // console.log("上传了：", fileItem.raw)
              } else {
                  throw new Error(res.msg || `文件 ${fileItem.name} 上传失败`)
              }
          }
          form.value.id ? await updateSuggestion(form.value) : await addSuggestion(form.value);
          proxy?.$modal.msgSuccess('操作成功');

      } catch (error) {
          ElMessage.error(error.message || '提交失败，请重试')
          // 如果有部分文件上传成功但业务提交失败，这里可以根据需求决定是否回滚（一般需要后端支持）
      } finally {
        submitting.value = false
      }

      dialog.visible = false;
      await getList();
    }
  });
};

// 统一的「能否关闭」判断：返回 true 才允许关闭
const confirmClose = async () => {
  const editor = editorRef.value
  // 没有未保存内容 → 直接关
  if (!editor || !editor.getIsDirty()) return true

  try {
    await ElMessageBox.confirm(
      '文档有未保存的更改，是否保存后再关闭？',
      '提示',
      {
        confirmButtonText: '保存并关闭',
        cancelButtonText: '不保存',
        distinguishCancelAndClose: true, // 关键：区分「取消按钮」与「关闭(X/ESC)」
        type: 'warning'
      }
    )
    // 点击了「保存并关闭」：必须等保存完成再关（destroy-on-close 会销毁编辑器）
    isSaving.value = true
    const ok = await editor.save()
    isSaving.value = false
    if (!ok) {
      ElMessage.error('保存失败，请检查回调服务后重试')
      return false
    }
    return true
  } catch (action) {
    // 'cancel' = 点了「不保存」→ 允许关闭；'close' = 点了 X/ESC → 留在当前页
    return action === 'cancel'
  }
}

// 点弹框 X / 按 ESC 触发（close-on-click-modal 已关，遮罩不会触发）
const handleBeforeClose = (done) => {
  if (isSaving.value) { ElMessage.warning('正在保存，请不要关闭'); return }
  confirmClose().then((canClose) => { if (canClose) done() })
}

// 底部「关闭」按钮
const handleClose = async () => {
  if (isSaving.value) { ElMessage.warning('正在保存，请不要关闭'); return }
  const canClose = await confirmClose()
  if (canClose) {
    dialogVisible.value = false
    mode.value = 'edit'
  }
}
